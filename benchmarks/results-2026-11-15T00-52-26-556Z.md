const fs = require('fs');
const path = require('path');

function isPathInside(projectRoot, requestedPath) {
  // Verifique se os caminhos são válidos
  if (!fs.existsSync(projectRoot) || (!requestedPath && requestedPath !== '') requestedPath = projectRoot, !fs.existsSync(requestedPath)) return false;

  // Resolva links simbólicos
  const resolvedPath = path.resolve(projectRoot, requestedPath);

  // Verifique se o caminho é um diretório chamado 'project-backup' ou um subdiretório chamado 'project'
  if (resolvedPath.includes('project-backup') || resolvedPath.includes('project')) return false;

  // Verifique se o caminho está dentro do diretório raiz autorizado
  const relativePath = path.relative(projectRoot, resolvedPath);
  return relativePath && !relativePath.startsWith('../') && !relativePath.startsWith('./');
}
